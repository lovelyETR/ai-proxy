【先理解一件事，后面就都通了】

  这个接口是「引擎无关」的。

  它就是一个普通的 HTTP 服务，说的是 OpenAI 兼容格式。
  所以「适配某个引擎」不需要为那个引擎改任何东西 ——
  只要那个引擎的插件「会发 HTTP 请求」，就能接。

  起源引擎 / Unity / 虚幻 / 寒霜 / GMod / FiveM / Minecraft /
  Godot / CryEngine / GameMaker / 自研引擎 …… 全都一样。

  下面按引擎给了可直接复制的代码。
  如果你的引擎不在列表里，直接看最后一节「通用 HTTP 契约」。


================================================================================
  接口契约（所有引擎都遵守这个）
================================================================================

  地址    http://127.0.0.1:8787/v1/chat/completions
  方法    POST
  头      Content-Type: application/json
          Authorization: Bearer <你的令牌>
          （有些客户端用 x-api-key: <你的令牌>，也支持）

  请求体（OpenAI 兼容）：

      {
        "model": "deepseek-chat",
        "messages": [
          {"role": "system", "content": "你是一个助手"},
          {"role": "user",   "content": "你好"}
        ],
        "max_tokens": 400,
        "temperature": 0.3,
        "stream": false
      }

  响应体：

      {
        "choices": [
          {"message": {"role": "assistant", "content": "你好！"}}
        ],
        "usage": {"prompt_tokens": 12, "completion_tokens": 5}
      }

  ★ 取答案就是取  choices[0].message.content

  ★ 让插件解析 JSON 太麻烦？在 system 里明确要求模型只输出 JSON，
    然后用字符串截取。见下面「解析简化技巧」。


================================================================================
  解析简化技巧（各语言通用）
================================================================================

  如果那个引擎没有 JSON 库，别硬解析。两个办法：

  办法一：让模型输出极简格式，用字符串切

      在 messages 里加一句：
          「你的回答必须只有一行，格式：SCORE|REASON
             SCORE 是 0 到 100 的整数，REASON 不超过 30 字。不要输出别的。」

      然后拿到的 content 直接按 | 切两半就完了。
      这是最省事、最不容易出错的做法，强烈推荐给引擎受限的插件。

  办法二：只判断关键词

      content 里包含 "YES" 就是通过，包含 "NO" 就是不通过。
      简单粗暴但够用。

  1.1  BepInEx / MelonLoader —— C#（推荐写法，不阻塞主线程）
--------------------------------------------------------------------------------

  using System;
  using System.Net.Http;
  using System.Text;
  using System.Threading.Tasks;

  public static class AiClient
  {
      // 指向你的代理，不是官方地址
      private const string BaseUrl = "http://127.0.0.1:8787/v1/chat/completions";
      private const string Token   = "my-token-1";          // 令牌，不是真 Key

      private static readonly HttpClient Http = new HttpClient
      {
          Timeout = TimeSpan.FromSeconds(60)
      };

      /// <summary>问一句，返回纯文本答案。失败返回 null。</summary>
      public static async Task<string> AskAsync(string system, string user)
      {
          // 用极简格式，省掉 JSON 解析
          var payload =
              "{\"model\":\"deepseek-chat\"," +
              "\"messages\":[" +
                "{\"role\":\"system\",\"content\":" + Json(system + "\n回答必须只有一行，格式：OK|理由") + "}," +
                "\"role\":\"user\",\"content\":" + Json(user) + "}" +
              "]," +
              "\"max_tokens\":200,\"temperature\":0.2}";

          using (var req = new HttpRequestMessage(HttpMethod.Post, BaseUrl))
          {
              req.Headers.Add("Authorization", "Bearer " + Token);
              req.Content = new StringContent(payload, Encoding.UTF8, "application/json");

              using (var resp = await Http.SendAsync(req).ConfigureAwait(false))
              {
                  string body = await resp.Content.ReadAsStringAsync().ConfigureAwait(false);
                  if (!resp.IsSuccessStatusCode)
                      return null;
                  return ExtractContent(body);
              }
          }
      }

      /// <summary>把字符串转义成 JSON 字符串字面量。</summary>
      private static string Json(string s)
      {
          if (s == null) return "\"\"";
          var sb = new StringBuilder("\"");
          foreach (char c in s)
          {
              switch (c)
              {
                  case '"':  sb.Append("\\\""); break;
                  case '\\': sb.Append("\\\\"); break;
                  case '\n': sb.Append("\\n");  break;
                  case '\r': sb.Append("\\r");  break;
                  case '\t': sb.Append("\\t");  break;
                  default:
                      if (c < 0x20) sb.Append("\\u").Append(((int)c).ToString("x4"));
                      else sb.Append(c);
                      break;
              }
          }
          return sb.Append('"').ToString();
      }

      /// <summary>从响应里抠出 choices[0].message.content。</summary>
      private static string ExtractContent(string json)
      {
          const string key = "\"content\":";
          int i = json.IndexOf(key, StringComparison.Ordinal);
          if (i < 0) return null;
          i += key.Length;
          while (i < json.Length && json[i] == ' ') i++;
          if (i >= json.Length || json[i] != '"') return null;
          i++;

          var sb = new StringBuilder();
          while (i < json.Length)
          {
              char c = json[i];
              if (c == '\\' && i + 1 < json.Length)
              {
                  char n = json[++i];
                  switch (n)
                  {
                      case 'n': sb.Append('\n'); break;
                      case 'r': sb.Append('\r'); break;
                      case 't': sb.Append('\t'); break;
                      case '"': sb.Append('"');  break;
                      case '\\': sb.Append('\\'); break;
                      case 'u':
                          if (i + 4 < json.Length)
                          {
                              sb.Append((char)Convert.ToInt32(json.Substring(i + 1, 4), 16));
                              i += 4;
                          }
                          break;
                      default: sb.Append(n); break;
                  }
              }
              else if (c == '"') break;
              else sb.Append(c);
              i++;
          }
          return sb.ToString();
      }
  }

      // 用极简格式，省掉 JSON 解析
      char body[1024];

          "{\"model\":\"deepseek-chat\",\"messages\":["
          "{\"role\":\"system\",\"content\":\"回答必须只有一行，格式：YES|理由\"},"
          "{\"role\":\"user\",\"content\":\"今天天气怎么样\"}"
          "],\"max_tokens\":100}");

      hc.Post(body, OnAiResponse);
      return Plugin_Handled;
  }

  void OnAiResponse(HTTPStatus status, const char[] data, any value, const char[] error)
  {
      if (status != HTTPStatus_OK)
      {
          PrintToServer("[AI] 请求失败: %s", error);
          return;
      }

      // 简单取 content 字段
      char content[512];
      if (ExtractJsonString(data, "content", content, sizeof(content)))
          PrintToServer("[AI] 回答: %s", content);
  }

  // 简易 JSON 字符串提取（够用就行）
  bool ExtractJsonString(const char[] json, const char[] key, char[] out, int maxlen)
  {
      char needle[64];
      Format(needle, sizeof(needle), "\"%s\":", key);

      int pos = StrContains(json, needle);
      if (pos < 0) return false;
      pos += strlen(needle);

      // 跳过空格
      while (json[pos] == ' ') pos++;
      if (json[pos] != '"') return false;
      pos++;

      int i = 0;
      while (json[pos] != '"' && json[pos] != '\0' && i < maxlen - 1)
      {
          if (json[pos] == '\\' && json[pos + 1] != '\0')
          {
              pos++;
              if (json[pos] == 'n') out[i++] = '\n';
              else out[i++] = json[pos];
          }
          else out[i++] = json[pos];
          pos++;
      }
      out[i] = '\0';
      return true;
  }

  -- 需要 luasocket 或类似库
  local http = require("socket.http")
  local ltn12 = require("ltn12")

  local function ask_ai(user_text)
      local req_body = string.format(
          '{"model":"deepseek-chat","messages":[' ..
          '{"role":"system","content":"回答必须只有一行，格式：YES|理由"},' ..
          '{"role":"user","content":"%s"}],"max_tokens":100}', user_text)

      local resp = {}
      local ok, code = http.request{
          url = "http://127.0.0.1:8787/v1/chat/completions",
          method = "POST",
          headers = {
              ["Content-Type"]  = "application/json",
              ["Authorization"] = "Bearer my-token-1",
              ["Content-Length"] = tostring(#req_body),
          },
          source = ltn12.source.string(req_body),
          sink   = ltn12.sink.table(resp),
      }

      if not ok then return nil, "请求失败" end
      return table.concat(resp)
  end

  3.1  C++（原生，最稳）

  #include "HttpModule.h"
  #include "Interfaces/IHttpRequest.h"
  #include "Interfaces/IHttpResponse.h"
  #include "Dom/JsonObject.h"
  #include "Serialization/JsonSerializer.h"
  #include "Serialization/JsonWriter.h"

  void UMyAiClient::Ask(const FString& Prompt)
  {
      // 拼 JSON
      TSharedPtr<FJsonObject> Root = MakeShareable(new FJsonObject);
      Root->SetStringField(TEXT("model"), TEXT("deepseek-chat"));
      Root->SetNumberField(TEXT("max_tokens"), 200);

      TArray<TSharedPtr<FJsonValue>> Msgs;
      {
          TSharedPtr<FJsonObject> M = MakeShareable(new FJsonObject);
          M->SetStringField(TEXT("role"), TEXT("system"));
          M->SetStringField(TEXT("content"),
              TEXT("回答必须只有一行，格式：YES|理由"));
          Msgs.Add(MakeShareable(new FJsonValueObject(M)));
      }
      {
          TSharedPtr<FJsonObject> M = MakeShareable(new FJsonObject);
          M->SetStringField(TEXT("role"), TEXT("user"));
          M->SetStringField(TEXT("content"), Prompt);
          Msgs.Add(MakeShareable(new FJsonValueObject(M)));
      }
      Root->SetArrayField(TEXT("messages"), Msgs);

      FString Body;
      {
          TSharedRef<TJsonWriter<>> W = TJsonWriterFactory<>::Create(&Body);
          FJsonSerializer::Serialize(Root.ToSharedRef(), W);
      }

      // 发请求
      TSharedRef<IHttpRequest> Req = FHttpModule::Get().CreateRequest();
      Req->SetURL(TEXT("http://127.0.0.1:8787/v1/chat/completions"));
      Req->SetVerb(TEXT("POST"));
      Req->SetHeader(TEXT("Content-Type"), TEXT("application/json"));
      Req->SetHeader(TEXT("Authorization"), TEXT("Bearer my-token-1"));
      Req->SetContentAsString(Body);

      Req->OnProcessRequestComplete().BindLambda(
          [](FHttpRequestPtr, FHttpResponsePtr Resp, bool bOk)
          {
              if (!bOk || !Resp.IsValid()) return;

              TSharedPtr<FJsonObject> Json;
              TSharedRef<TJsonReader<>> R =
                  TJsonReaderFactory<>::Create(Resp->GetContentAsString());

              if (FJsonSerializer::Deserialize(R, Json) && Json.IsValid())
              {
                  const TArray<TSharedPtr<FJsonValue>>* Choices;
                  if (Json->TryGetArrayField(TEXT("choices"), Choices) && Choices->Num() > 0)
                  {
                      TSharedPtr<FJsonObject> Msg =
                          (*Choices)[0]->AsObject()->GetObjectField(TEXT("message"));
                      FString Content = Msg->GetStringField(TEXT("content"));
                      UE_LOG(LogTemp, Warning, TEXT("[AI] %s"), *Content);
                  }
              }
          });

      Req->ProcessRequest();
  }

--------------------------------------------------------------------------------
  3.2  蓝图（Blueprint）
--------------------------------------------------------------------------------

  蓝图没有原生 HTTP 节点，需要装插件。常见选择：

      VaRest       —— 最常用，免费，节点齐全
      HttpBlueprint —— 也有
      RuntimeHttp  —— 轻量

  以 VaRest 为例，节点连法：

      1. Construct Json Object
            加字段 model   = "deepseek-chat"
            加字段 messages（数组）
              每个元素是一个 Json Object：
                role    = "system" / "user"
                content = 文本

      2. Call URL
            Verb    = POST
            URL     = http://127.0.0.1:8787/v1/chat/completions
            Headers = Content-Type: application/json
                      Authorization: Bearer my-token-1
            Body    = 上面构造的 Json Object

      3. 绑定 On Request Complete / On Request Fail
      4. 从返回的 Json 里取 choices[0].message.content

  ★ VaRest 的 Get String Field 节点可以直接取字段，不用自己解析。

      local TOKEN = "my-token-1"

      local function AskAI(userText, onDone)
          local payload = util.TableToJSON({
              model = "deepseek-chat",
              messages = {
                  { role = "system", content = "回答必须只有一行，格式：YES|理由" },
                  { role = "user",   content = userText },
              },

          })

          http.Post("http://127.0.0.1:8787/v1/chat/completions", payload, function(body, size, headers, code)
              if code ~= 200 then
                  print("[AI] 请求失败，HTTP " .. tostring(code))
                  return
              end

              local obj = util.JSONToTable(body)
              if not obj or not obj.choices then return end

              local answer = obj.choices[1].message.content
              if onDone then onDone(answer) end
          end, {
              ["Content-Type"]  = "application/json",
              ["Authorization"] = "Bearer " .. TOKEN,
          })
      end

      -- 用法
      AskAI("这个玩家行为正常吗", function(ans)
          print("[AI] " .. ans)
      end)

--------------------------------------------------------------------------------
  7.1  Forge / Fabric 模组（Java 11+）

  import java.net.URI;
  import java.net.http.HttpClient;
  import java.net.http.HttpRequest;
  import java.net.http.HttpResponse;
  import java.time.Duration;

  public final class AiClient {

      private static final String URL   = "http://127.0.0.1:8787/v1/chat/completions";
      private static final String TOKEN = "my-token-1";

      private static final HttpClient HTTP = HttpClient.newBuilder()
              .connectTimeout(Duration.ofSeconds(10))
              .build();

      /** 问一句，返回纯文本。失败返回 null。 */
      public static String ask(String userText) {
          String body = "{"
              + "\"model\":\"deepseek-chat\","
              + "\"messages\":["
              +   "{\"role\":\"system\",\"content\":\"回答必须只有一行，格式：YES|理由\"},"
              +   "{\"role\":\"user\",\"content\":" + json(userText) + "}"
              + "],"
              + "\"max_tokens\":100"
              + "}";

          HttpRequest req = HttpRequest.newBuilder()
                  .uri(URI.create(URL))
                  .timeout(Duration.ofSeconds(60))
                  .header("Content-Type", "application/json")
                  .header("Authorization", "Bearer " + TOKEN)
                  .POST(HttpRequest.BodyPublishers.ofString(body))
                  .build();

          try {
              HttpResponse<String> resp = HTTP.send(req, HttpResponse.BodyHandlers.ofString());
              if (resp.statusCode() != 200) return null;
              return extractContent(resp.body());
          } catch (Exception e) {
              return null;
          }
      }

      /** 别在主线程里同步调用 —— 用线程池或 CompletableFuture。 */
      public static void askAsync(String userText, java.util.function.Consumer<String> cb) {
          java.util.concurrent.CompletableFuture
              .supplyAsync(() -> ask(userText))
              .thenAccept(cb);
      }

      private static String json(String s) {
          StringBuilder sb = new StringBuilder("\"");
          for (char c : s.toCharArray()) {
              switch (c) {
                  case '"':  sb.append("\\\""); break;
                  case '\\': sb.append("\\\\"); break;
                  case '\n': sb.append("\\n");  break;
                  case '\r': sb.append("\\r");  break;
                  case '\t': sb.append("\\t");  break;
                  default:
                      if (c < 0x20) sb.append(String.format("\\u%04x", (int) c));
                      else sb.append(c);
              }
          }
          return sb.append('"').toString();
      }

      private static String extractContent(String json) {
          String key = "\"content\":";
          int i = json.indexOf(key);
          if (i < 0) return null;
          i += key.length();
          while (i < json.length() && json.charAt(i) == ' ') i++;
          if (i >= json.length() || json.charAt(i) != '"') return null;
          i++;

          StringBuilder sb = new StringBuilder();
          while (i < json.length()) {
              char c = json.charAt(i);
              if (c == '\\' && i + 1 < json.length()) {
                  char n = json.charAt(++i);
                  switch (n) {
                      case 'n':  sb.append('\n'); break;
                      case 'r':  sb.append('\r'); break;
                      case 't':  sb.append('\t'); break;
                      case '"':  sb.append('"');  break;
                      case '\\': sb.append('\\'); break;
                      case 'u':
                          if (i + 4 < json.length()) {
                              sb.append((char) Integer.parseInt(json.substring(i + 1, i + 5), 16));
                              i += 4;
                          }
                          break;
                      default: sb.append(n);
                  }
              } else if (c == '"') {
                  break;
              } else {
                  sb.append(c);
              }
              i++;
          }
          return sb.toString();
      }
  }

  import json
  import urllib.request

  def ask_ai(user_text, system="回答必须只有一行，格式：YES|理由"):
      body = json.dumps({
          "model": "deepseek-chat",
          "messages": [
              {"role": "system", "content": system},

          ],
          "max_tokens": 100,
          "temperature": 0.2,
      }).encode("utf-8")

      req = urllib.request.Request(
          "http://127.0.0.1:8787/v1/chat/completions",

          headers={
              "Content-Type": "application/json",
              "Authorization": "Bearer my-token-1",

          with urllib.request.urlopen(req, timeout=60) as r:
              obj = json.loads(r.read().decode("utf-8"))
              return obj["choices"][0]["message"]["content"]
      except Exception as e:
          print("[AI] 请求失败:", e)
          return None


  if __name__ == "__main__":
      print(ask_ai("今天天气怎么样"))

================================================================================
  十、排查顺序（照这个走，99% 的问题都能定位）
================================================================================

  第 1 步：接口本身活着吗

      curl http://127.0.0.1:8787/health

      返回 {"ok": true, ...} 就是活的。
      连不上 → 代理没启动，或端口被占。

  第 2 步：模型列表能拿到吗

      curl http://127.0.0.1:8787/v1/models

      这一步是给「插件启动时探测模型」用的。
      如果你的插件启动就报错，八成是这个接口没通。

  第 3 步：带令牌能请求通吗

      用 9.3 那条 curl。
      返回 401 → 令牌填错了（要填 AI_TOKENS 里的，不是真 Key）。
      返回 429 → 触发限流了，把 AI_RATE_PER_MIN 调大。
      返回 500 → AI_API_KEY 没填。
      返回 502 → 上游连不上（检查 AI_UPSTREAMS 和网络）。

  第 4 步：插件里还是不行？

      - 看服务端黑窗口有没有日志（代理会打印每次请求）
      - 确认插件里的 api_base_url 结尾有没有 /v1
      - 确认请求头带的是 Bearer（不是别的写法）
      - 有些引擎对 127.0.0.1 有限制，试试本机局域网 IP

  第 5 步：看用量

      curl -H "Authorization: Bearer <AI_ADMIN_TOKEN>" http://127.0.0.1:8787/stats

      能看到每个令牌用了多少次、多少 token。
      请求数是 0 → 插件根本没发出来（是插件那边的问题，不是接口）。
